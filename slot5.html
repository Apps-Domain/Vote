        # Count the number of vowels, consonants, digits and special characters
str=input("Enter a String:") #Reads the string
vowels=consonants=digits=specialchar=0

for char in str:
    if char in 'aeiouAEIOU': vowels+=1
    if char in 'bBcCdDfFgGhHjJkKlLmMnNpPqQrRsStTvVwWxXyYzZ': consonants+=1
    if char in '0123456789': digits+=1
    if char in '!@#$%^&*<>?:': specialchar+=1

print("The number of vowels=",vowels)
print("The number of consonants=",consonants)
print("The number of digits=",digits)
print("The number of special characters=",specialchar)

        # Sum of all elements in a list
